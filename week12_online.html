끄고 줄일 수 있게 — 설정 두 칸

흔들림 끄기 · 소리 크기. 토글과 슬라이더에 한 줄씩.

GameSettings.cs — 설정 화면의 빈 오브젝트에
using UnityEngine;

public class GameSettings : MonoBehaviour
{
    public void SetShake(bool on)       // 토글 → On Value Changed
    {
        Feedback.shakeOn = on;
    }

    public void SetVolume(float value)  // 슬라이더 → On Value Changed
    {
        AudioListener.volume = value;   // 0 ~ 1, 게임 전체 소리
    }
}

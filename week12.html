using System.Collections;
using UnityEngine;

// 빈 오브젝트 FeelKit 에 붙이고 cam · sfx 칸을 채워요
public class FeelKit : MonoBehaviour
{
    public static FeelKit I;      // 어디서든 FeelKit.I.HitStop(0.06f)
    public Transform cam;         // 빈 오브젝트 CameraRig 아래 Main Camera
    public AudioSource sfx;       // 효과음용 AudioSource
    Vector3 camHome;
    float shake;                  // 남은 흔들림 세기

    void Awake() { I = this; camHome = cam.localPosition; }

    public void HitStop(float sec) { StartCoroutine(Stop(sec)); }
    IEnumerator Stop(float sec)
    {
        Time.timeScale = 0f;
        yield return new WaitForSecondsRealtime(sec);
        Time.timeScale = 1f;
    }

    public void Shake(float power) { shake = Mathf.Max(shake, power); }
    void LateUpdate()
    {
        cam.localPosition = camHome + (Vector3)(Random.insideUnitCircle * shake);
        shake = Mathf.MoveTowards(shake, 0f, Time.unscaledDeltaTime * 2f);
    }

    public void Play(AudioClip clip)
    {
        sfx.pitch = Random.Range(0.9f, 1.1f);   // 같은 소리도 조금씩 다르게
        sfx.PlayOneShot(clip);
    }
}

확인 기준 1 · 베타 한 문장 — 알파와 같은 틀

「번에서 이(가) 해 본 사람의 이면 확인된 것으로 본다.」

예) 1번에서 4 ~ 5 가 해 본 사람의 절반 이상이면 확인된 것으로 본다.

확인 기준 2 · 고친 것 — 알파 숫자와 나란히

「알파에서 이(가) 명 중 명이었다. 베타 번에서 이(가) 이하이면 고친 것이 통한 것으로 본다.」

예) 알파에서 「적이 움직이는 줄 몰랐다」가 15명 중 5명이었다. 베타 4번에서 「한 번 닿고 나서 · 끝까지 몰랐다」가 해 본 사람의 5분의 1 이하이면 고친 것이 통한 것으로 본다.

둘 다 폼에는 넣지 않고 과제 글에만 써요. 베타가 끝나면 이 두 줄에 대어 셉니다

씬을 넘기는 SceneLoader 스크립트

버튼이 부를 함수 셋을 한 파일에 둬요. 함수는 public void 여야 버튼 목록에 나와요

Assets/Scripts/SceneLoader.cs — 파일 이름 = class 이름
using UnityEngine; using UnityEngine.SceneManagement; public class SceneLoader : MonoBehaviour { public void LoadGame() // START · RETRY { Time.timeScale = 1f; // 멈춘 시간을 먼저 되돌린다 SceneManager.LoadScene("Game"); } public void LoadTitle() // TITLE { Time.timeScale = 1f; SceneManager.LoadScene("Title"); } public void Quit() // QUIT { Application.Quit(); // 빌드에서만 꺼진다 } }
"Game" 은 씬 파일 이름 그대로

대문자 · 띄어쓰기까지 같게. 씬 이름을 바꾸면 여기도 같이 바꿔요

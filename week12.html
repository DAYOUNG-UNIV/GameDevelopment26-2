using System.Collections;
using UnityEngine;

// 빈 오브젝트 FeelKit 에 붙이고 cam · sfx 칸을 채워요
public class FeelKit : MonoBehaviour
{
    public static FeelKit I;      // 어디서든 FeelKit.I.HitStop(0.06f)
    public Transform cam;         // 빈 오브젝트 CameraRig 아래 Main Camera
    public AudioSource sfx;       // 효과음용 AudioSource
    Vector3 camHome;
    float shake;                  // 남은 Shake 세기

    void Awake() { I = this; camHome = cam.localPosition; }

    public void HitStop(float sec) { StartCoroutine(Stop(sec)); }
    IEnumerator Stop(float sec)
    {
        Time.timeScale = 0f;
        yield return new WaitForSecondsRealtime(sec);
        Time.timeScale = 1f;
    }

    public void Shake(float power) { shake = Mathf.Max(shake, power); }
    void LateUpdate()
    {
        cam.localPosition = camHome + (Vector3)(Random.insideUnitCircle * shake);
        shake = Mathf.MoveTowards(shake, 0f, Time.unscaledDeltaTime * 2f);
    }

플레이어가 죽거나 클리어할 때 자리와 시간을 한 줄씩 적어요. 베타가 끝나면 시트에 붙여 지도로 봐요

DeathLog.cs
using System.Globalization;
using System.IO;
using UnityEngine;
using UnityEngine.SceneManagement;

// Game 씬의 빈 오브젝트 'DeathLog' 에 붙여요.
// 플레이어가 죽는 곳에서  DeathLog.Dead(transform.position);
// 클리어하는 곳에서      DeathLog.Clear(transform.position);
public class DeathLog : MonoBehaviour
{
    static string path;
    static float startTime;

    void Awake()
    {
        path = Path.Combine(Application.persistentDataPath, "beta_log.csv");
        if (!File.Exists(path)) File.WriteAllText(path, "time,scene,event,x,y,sec\n");
        startTime = Time.realtimeSinceStartup;   // 씬을 다시 불러오면 0 부터
    }

    public static void Dead(Vector3 p)  { Write("dead", p); }
    public static void Clear(Vector3 p) { Write("clear", p); }

    static void Write(string ev, Vector3 p)
    {
        if (path == null) return;   // DeathLog 오브젝트가 씬에 없으면 그냥 넘어가요
        var c = CultureInfo.InvariantCulture;
        string line = System.DateTime.Now.ToString("HH:mm:ss") + ","
                    + SceneManager.GetActiveScene().name + "," + ev + ","
                    + p.x.ToString("F1", c) + "," + p.y.ToString("F1", c) + ","
                    + (Time.realtimeSinceStartup - startTime).ToString("F1", c) + "\n";
        File.AppendAllText(path, line);
    }
}
